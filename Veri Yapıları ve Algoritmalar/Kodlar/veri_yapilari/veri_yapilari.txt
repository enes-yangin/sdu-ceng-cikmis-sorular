void tersCevir(struct node** x) {
	struct node *s, *q, *r;
	q = *x;
	r = NULL;
	while (q != NULL) {
		s = r;
		r = q;
		q = q->next;
		r->next = s;
	}
	*x = r;
}

void basaEkle(struct node **head, int x){
	struct node *ptr = (struct node*)malloc(sizeof(struct node));
	ptr->data=x;
	ptr->prev=NULL;
	ptr->next=(*head);
	(*head)->prev=ptr;
	(*head)=ptr;
}

void arayaEkle(struct node *head, int x, int y){
	struct node *yeni= (struct node*)malloc(sizeof(struct node));
	yeni->data=x;
	struct node *ptr=head;
	while(ptr!=NULL){
		if(ptr->data==y)
			break;
		ptr=ptr->next;
	}
	if(ptr==NULL)
		return;
	yeni->next=ptr->next;
	yeni->prev=ptr;
	ptr->next->prev=yeni;
	ptr->next=yeni;
}
